CREATE TABLE ingredientes (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    quantidade INT NOT NULL
);


CREATE TABLE consumo (
    id SERIAL PRIMARY KEY,
    id_ingrediente INT NOT NULL,
    quantidade_utilizada INT NOT NULL,
    CONSTRAINT fk_ingrediente FOREIGN KEY (id_ingrediente) REFERENCES ingredientes(id)
);


INSERT INTO ingredientes (id, nome, quantidade) VALUES
(1, 'Pão de Hambúrguer', 150),
(2, 'Hambúrguer Bovino', 120),
(3, 'Queijo Cheddar', 100),
(4, 'Alface', 80),
(5, 'Tomate', 90);


INSERT INTO consumo (id, id_ingrediente, quantidade_utilizada) VALUES
(1, 1, 10),
(2, 2, 10),
(3, 3, 8);


CREATE OR REPLACE FUNCTION fn_atualiza_ingredientes()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE ingredientes
    SET quantidade = quantidade - NEW.quantidade_utilizada
    WHERE id = NEW.id_ingrediente;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER trg_atualiza_ingredientes
AFTER INSERT ON consumo
FOR EACH ROW
EXECUTE FUNCTION fn_atualiza_ingredientes();


CREATE OR REPLACE FUNCTION fn_lista_consumo()
RETURNS TABLE (
    nome_ingrediente VARCHAR,
    quantidade_restante INT,
    quantidade_utilizada INT,
    quantidade_original INT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        i.nome::VARCHAR,
        i.quantidade AS quantidade_restante,
        c.quantidade_utilizada,
        (i.quantidade + c.quantidade_utilizada) AS quantidade_original
    FROM consumo c
    JOIN ingredientes i ON c.id_ingrediente = i.id;
END;
$$ LANGUAGE plpgsql;


INSERT INTO consumo (id_ingrediente, quantidade_utilizada) VALUES (5, 5);


SELECT * FROM fn_lista_consumo();

    nome_cliente VARCHAR,
    saldo_atual NUMERIC,
    valor_saque NUMERIC,

) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        c.cliente::VARCHAR,
        c.saldo AS saldo_atual,
        s.valor_saque,
        (c.saldo + s.valor_saque) AS saldo_antes
    FROM saques s
    JOIN contas c ON s.id_conta = c.id;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_relatorio_pedidos()
RETURNS TABLE (
    nome_cliente VARCHAR,
    nome_produto VARCHAR,
    quantidade_comprada INT,
    estoque_restante INT,
    estoque_original INT
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        cl.nome::VARCHAR,
        p.nome::VARCHAR,
        pd.quantidade,
        p.estoque,
        (p.estoque + pd.quantidade)
    FROM pedidos pd
    JOIN clientes cl ON pd.id_cliente = cl.id
    JOIN produtos p ON pd.id_produto = p.id;
END;
$$ LANGUAGE plpgsql;
